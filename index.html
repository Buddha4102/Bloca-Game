} } // Afficher hint UIManager ui = GameManager.Instance?.uiManager; if (_nearestNPC != null) ui?.ShowInteractHint(_nearestNPC.npcName); else ui?.HideInteractHint(); // Déclencher dialogue if (_nearestNPC != null && (Input.GetKeyDown(interactKey) || IsInteractButtonPressed())) { _nearestNPC.StartDialogue();
